cmake_minimum_required(VERSION 3.16)
project(cybr_light VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP)

function(cybr_options target optimization suppress_unused)
  target_include_directories(${target} PRIVATE include)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /W4)
    target_compile_definitions(${target} PRIVATE NOMINMAX)
    if(suppress_unused)
      target_compile_options(${target} PRIVATE /wd4100)
    endif()
  else()
    target_compile_options(${target} PRIVATE ${optimization} -Wall -Wextra)
    if(suppress_unused)
      target_compile_options(${target} PRIVATE -Wno-unused-parameter)
    endif()
  endif()
  target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
endfunction()

add_executable(cybr-light src/main.cpp)
cybr_options(cybr-light -O3 TRUE)
add_executable(cybr-photons src/photons.cpp)
cybr_options(cybr-photons -O3 TRUE)
foreach(target cybr-light cybr-photons)
  # Classic MSVC OpenMP rejects the photon detector's uint64 loop index.
  # Keep its original computation serial on MSVC; GCC/Clang retain OpenMP.
  if(OpenMP_CXX_FOUND AND (NOT MSVC OR target STREQUAL "cybr-light"))
    target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endforeach()

enable_testing()
add_executable(cybr-tests tests/numerical.cpp)
cybr_options(cybr-tests -O2 FALSE)
add_test(NAME numerical COMMAND cybr-tests)
add_executable(cybr-engine-tests tests/engine_numerical.cpp)
cybr_options(cybr-engine-tests -O2 TRUE)
add_test(NAME engine_numerical COMMAND cybr-engine-tests)
